cmake_minimum_required(VERSION 3.24)
project(OpenGridXR VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(opengrid_core STATIC src/settings.cpp third_party/horizonlock/core.cpp)
target_include_directories(opengrid_core PUBLIC include third_party/horizonlock third_party/openxr)
target_compile_definitions(opengrid_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_executable(opengrid-core-tests tests/core_tests.cpp)
target_link_libraries(opengrid-core-tests PRIVATE opengrid_core)
add_test(NAME native-core COMMAND opengrid-core-tests)
if(WIN32)
 add_library(opengrid_helmet STATIC src/helmet_renderer.cpp src/helmet_math.cpp telemetry/telemetry.cpp)
 target_include_directories(opengrid_helmet PUBLIC include third_party/openxr telemetry)
 target_compile_definitions(opengrid_helmet PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(opengrid_helmet PUBLIC d3d11 d3d12 dxgi windowscodecs ole32)
 add_executable(opengrid-helmet-math-tests tests/helmet_math_tests.cpp)
 target_link_libraries(opengrid-helmet-math-tests PRIVATE opengrid_helmet)
 add_test(NAME native-helmet-math COMMAND opengrid-helmet-math-tests)
 add_executable(opengrid-helmet-renderer-tests tests/helmet_renderer_tests.cpp)
 target_link_libraries(opengrid-helmet-renderer-tests PRIVATE opengrid_helmet)
 add_test(NAME native-helmet-d3d11 COMMAND opengrid-helmet-renderer-tests)
 add_test(NAME native-helmet-d3d11on12 COMMAND opengrid-helmet-renderer-tests --d3d12)
 add_library(OpenGridXR SHARED src/layer.cpp src/image_processing.cpp)
 target_link_libraries(OpenGridXR PRIVATE opengrid_core d3d11 dxgi d3dcompiler)
 add_executable(OpenGridXR-Output WIN32 src/output.cpp)
 target_link_libraries(OpenGridXR-Output PRIVATE opengrid_core d3d11 dxgi d3dcompiler user32 gdi32 shell32)
 add_custom_command(TARGET OpenGridXR POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/OpenGridXR.json" "$<TARGET_FILE_DIR:OpenGridXR>/OpenGridXR.json")
 add_test(NAME native-output-startup COMMAND OpenGridXR-Output --self-test --game iracing)
 add_executable(opengrid-layer-tests tests/layer_tests.cpp)
 set_target_properties(opengrid-layer-tests PROPERTIES OUTPUT_NAME "iRacingSim64DX11_test")
 target_link_libraries(opengrid-layer-tests PRIVATE opengrid_core d3d11 dxgi)
 add_dependencies(opengrid-layer-tests OpenGridXR)
 add_test(NAME native-layer COMMAND opengrid-layer-tests)
 add_library(opengrid-engine-fixture SHARED tests/engine_fixture.cpp)
 target_link_libraries(opengrid-engine-fixture PRIVATE opengrid_core)
 add_test(NAME native-private-chain COMMAND opengrid-layer-tests --engine $<TARGET_FILE:opengrid-engine-fixture>)
 add_test(NAME native-private-disabled COMMAND opengrid-layer-tests --engine-disabled $<TARGET_FILE:opengrid-engine-fixture>)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/Release/OpenGridXR-Enhancements.dll")
  add_test(NAME native-private-engine COMMAND opengrid-layer-tests --real-engine "${CMAKE_CURRENT_SOURCE_DIR}/build/$<CONFIG>/OpenGridXR-Enhancements.dll")
 endif()
 add_executable(opengrid-gpu-tests tests/gpu_tests.cpp)
 target_link_libraries(opengrid-gpu-tests PRIVATE opengrid_core d3d11 dxgi d3dcompiler user32 gdi32 shell32)
 add_test(NAME native-gpu COMMAND opengrid-gpu-tests)
 add_executable(opengrid-capture-tests tests/capture_tests.cpp src/image_processing.cpp)
 target_link_libraries(opengrid-capture-tests PRIVATE opengrid_core d3d11 dxgi d3dcompiler)
 add_test(NAME native-capture COMMAND opengrid-capture-tests)
 add_executable(opengrid-image-processing-tests tests/image_processing_tests.cpp src/image_processing.cpp)
 target_link_libraries(opengrid-image-processing-tests PRIVATE opengrid_core d3d11 dxgi d3dcompiler)
 add_test(NAME native-image-processing COMMAND opengrid-image-processing-tests)
endif()
